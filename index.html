7เปิด Validation และ Swagger ใน main.ts

import { NestFactory } from '@nestjs/core';
import { ValidationPipe } from '@nestjs/common';
import { DocumentBuilder, SwaggerModule } from '@nestjs/swagger';
import { AppModule } from './app.module';

async function bootstrap() {
  const app = await NestFactory.create(AppModule);
  app.useGlobalPipes(new ValidationPipe());
  const cfg = new DocumentBuilder().setTitle('Library API').build();
  SwaggerModule.setup('docs', app, SwaggerModule.createDocument(app, cfg));
  await app.listen(3000);
}
bootstrap();

การใช้ Swagger

Swagger ทำอะไรได้

สร้างหน้าเอกสาร API ให้อัตโนมัติจากโค้ด เปิดที่ /docs เห็นทุก endpoint และกด Try it out เพื่อยิง API ทดสอบได้เลย (ตั้งค่าใน main.ts ตามขั้นที่ 7 ด้านบน)

DTO ใส่ @ApiProperty({ example: ... }) เพื่อให้ช่อง Request body มีตัวอย่างข้อมูล

Controller ใส่ decorator เหล่านี้เพื่อบอกสถานะที่ตอบกลับ

import { ApiTags, ApiOperation, ApiOkResponse,
  ApiCreatedResponse, ApiNotFoundResponse } from '@nestjs/swagger';

@ApiTags('books')            // จัดกลุ่มหัวข้อ
@Controller('books')
export class BooksController {
  @Get(':id')
  @ApiOperation({ summary: 'ดึงหนังสือเล่มเดียว' })
  @ApiOkResponse({ description: 'พบหนังสือ (200)' })
  @ApiNotFoundResponse({ description: 'ไม่พบหนังสือ (404)' })
  findOne(@Param('id', ParseIntPipe) id: number) { ... }

  @Post()
  @ApiCreatedResponse({ description: 'สร้างสำเร็จ (201)' })
  create(@Body() dto: BookDto) { ... }
}

ถ้าอยากไม่ต้องเขียน @ApiProperty ทุกฟิลด์ ให้เปิด CLI plugin โดยเพิ่ม @nestjs/swagger ใน nest-cli.json (ดูลิงก์หัวข้อ “เรื่อง Swagger” ท้ายหน้า)

2Gen โครง NestJS จาก OpenAPI

ตัวเลือก A: @snow-tzu/nest-openapi-code-generator แบบ contract-first สร้าง controller, DTO (มี class-validator และ @ApiProperty ให้) จาก OpenAPI 3.1

cd library-api
npm install @snow-tzu/nest-openapi-code-generator
mkdir specs
# วางไฟล์ spec ไว้ที่ specs/books.openapi.yaml
npx openapi-generate --specs ./specs --output ./src/generated

ตัวสร้างมีโหมด --watch ที่ gen ใหม่เมื่อไฟล์ spec เปลี่ยน และตั้งค่าเพิ่มผ่าน openapi.config.ts ได้ (specsDir, outputDir, generateControllers, generateDtos) ชื่อคลาสที่ได้อิงจากชื่อไฟล์ spec จึงควรตั้งชื่อเป็น books.openapi.yaml

ตัวเลือก B: @fresha/openapi-codegen สร้าง module, controller, action และ DTO ตามแนวทางของ NestJS

npm install @fresha/openapi-codegen

ข้อควรระวัง: generator typescript-nestjs ของ OpenAPI Generator (openapi-generator-cli) เป็น ฝั่ง client และยังเป็น experimental ไม่ใช่ตัวสร้างโครง server ดังนั้นสำหรับ backend ให้ใช้ตัวเลือก A หรือ B

อาการสาเหตุและวิธีแก้
fetch failed / ECONNREFUSEDBackend ยังไม่รัน หรือ NEXT_PUBLIC_API_URL ผิด (แก้ .env.local แล้วต้องรีสตาร์ต npm run dev)
CORS error ใน browserลืม enableCors ใน main.ts หรือยังไม่รีสตาร์ต Nest
กดบันทึกแล้วได้ 400ข้อมูลไม่ผ่าน validation เช่น ชื่อว่าง หรือราคาติดลบ
Cannot find module '@/lib/api'โฟลเดอร์ lib ต้องอยู่ที่รากโปรเจกต์ (ไม่ใช่ใน app) และ import alias ต้องเป็น @/*

⚡ ฟังก์ชันของ Next.js ที่ใช้บ่อย

กดหัวข้อเพื่อดูฟังก์ชัน ตัวอย่าง และผลลัพธ์ อ้างอิงจาก App Router ถ้าใช้คนละเวอร์ชัน (บางส่วนเปลี่ยนใน Next 15 และ 16) ให้เทียบกับ nextjs.org/docs

📘 ฟังก์ชันจัดการตัวแปรแต่ละชนิดใน TypeScript

กดหัวข้อเพื่อเปิดดูตาราง ฟังก์ชัน ตัวอย่าง และผลลัพธ์ แต่ละแถวเป็นอิสระต่อกัน (ไม่ต่อเนื่องจากแถวก่อนหน้า) บางฟังก์ชัน เช่น replaceAll, at, Promise.any ต้องตั้ง target หรือ lib ใน tsconfig เป็น ES2022 ขึ้นไป (โปรเจกต์ NestJS รุ่นใหม่ตั้งไว้ให้แล้ว)
